import { Bar, BarChart, XAxis, YAxis } from 'recharts';
import { CheckCircle, XCircle, Clock } from 'lucide-react';

import {
  Card,
  CardContent,
  CardFooter,
  CardHeader,
  CardTitle
} from '@/components/ui/card';
import {
  ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent
} from '@/components/ui/chart';
import { QUERY_RESULT } from '@/hooks/useCustomQuery';
import { ATTENDANCE_LIST_STUDENT } from './type';

export const description = 'Attendance Chart List';

type AttendanceChartListProps = {
  attendanceData: QUERY_RESULT<ATTENDANCE_LIST_STUDENT>;
};

export function AttendanceChartList({
  attendanceData
}: Readonly<AttendanceChartListProps>) {
  const attendanceChartListData = attendanceData.data?.data?.percentageData
    ?.map((item: any) => {
      if (item.name === 'Present') {
        return {
          browser: 'Present',
          visitors: item.value,
          fill: '#10b981'
        };
      } else if (item.name === 'Absent') {
        return {
          browser: 'Absent',
          visitors: item.value,
          fill: '#ef4444'
        };
      } else if (item.name === 'Leave') {
        return {
          browser: 'Leave',
          visitors: item.value,
          fill: '#f59e0b'
        };
      }
      return null;
    })
    .filter(Boolean);

  const chartConfig = {
    visitors: {
      label: 'Attendance',
      color: 'hsl(var(--chart-1))'
    },
    Present: {
      label: 'Present',
      color: '#10b981'
    },
    Absent: {
      label: 'Absent',
      color: '#ef4444'
    },
    Leave: {
      label: 'Leave',
      color: '#f59e0b'
    }
  } as ChartConfig;

  return (
    <Card className="shadow-md border-0">
      <CardHeader className="pb-2">
        <CardTitle className="text-lg font-bold bg-gradient-to-r from-blue-600 to-purple-600 bg-clip-text text-transparent">
          Attendance Summary
        </CardTitle>
      </CardHeader>
      <CardContent>
        <ChartContainer config={chartConfig}>
          <BarChart
            accessibilityLayer
            data={attendanceChartListData}
            layout="vertical"
            margin={{
              left: 0,
              right: 0,
              top: 10,
              bottom: 10
            }}
            barSize={24}
          >
            <YAxis
              dataKey="browser"
              type="category"
              tickLine={false}
              tickMargin={8}
              axisLine={false}
              width={60}
              tick={{ fontSize: 14, fontWeight: 500 }}
              tickFormatter={(value) =>
                chartConfig[value as keyof typeof chartConfig]?.label as string
              }
            />
            <XAxis dataKey="visitors" type="number" hide />
            <ChartTooltip cursor={false} content={<ChartTooltipContent />} />
            <Bar dataKey="visitors" layout="vertical" radius={4} />
          </BarChart>
        </ChartContainer>
      </CardContent>
      <CardFooter className="flex justify-between items-center pt-0 text-sm border-t">
        <div className="flex gap-4">
          <div className="flex items-center">
            <CheckCircle className="h-4 w-4 text-green-600 mr-1" />
            <span className="font-medium">Present</span>
          </div>
          <div className="flex items-center">
            <XCircle className="h-4 w-4 text-red-600 mr-1" />
            <span className="font-medium">Absent</span>
          </div>
          <div className="flex items-center">
            <Clock className="h-4 w-4 text-yellow-600 mr-1" />
            <span className="font-medium">Leave</span>
          </div>
        </div>
      </CardFooter>
    </Card>
  );
}
