import { AlertCircle, Edit, FileText } from 'lucide-react';
import { Card, CardContent } from '@/components/ui/card';
import { Link } from 'react-router-dom';
import Page from '@/components/helmet-page';
import { useTranslate } from '@/hooks/useTanslate';
import { useApi } from '@/hooks/useCustomQuery';
import { BASE_URI, feeDemandApi, timeTableApi } from '@/lib';
import { STUDENT_FEE_TYPE_LIST } from '../fee/payment/type';
import { useAuth } from '@/store/useAuth';
import { Button } from '@/components/ui/button';
import { Image } from '@/components/image';
import moment from 'moment';
import { useCircularEventList } from '@/hooks/useMaster';
import CircularEventById from '@/components/circular-event-by-id';
import { useState } from 'react';
import SearchMenu from './SearchBox';
import { menuItems } from './menuItem';
import StudentPortalSkeleton from './SkeletonLoader';
import { TIME_TABLE_LIST_TYPE } from '../time-table/type';
import { useFeeAmountByStudentId } from '@/hooks/useCommonData';

export default function Home() {
  const { user, sessionId } = useAuth();
  const [isDialogOpen, setIsDialogOpen] = useState<boolean>(false);
  const [getId, setId] = useState<string | null>(null);
  const circularEventData = useCircularEventList({
    classId: user?.student?.classId!
  });
  const { i18n } = useTranslate();
  const currentMonthFull = moment().format('MMMM');
  const feeAmountByStudentId = useFeeAmountByStudentId(
    user?.student?._id!,
    currentMonthFull
  );

  const classData = useApi<TIME_TABLE_LIST_TYPE>({
    api: `${timeTableApi.getTimeTable}?sessionId=${sessionId}`,
    key: 'timeTable',
    value: [sessionId],
    options: {
      enabled: true
    }
  });

  const feeDemandData = useApi<STUDENT_FEE_TYPE_LIST>({
    api: `${feeDemandApi.feeDemandByStudentIdStatics}?sessionId=${sessionId}`,
    options: {
      enabled: true
    }
  });

  const getIsPaidAmount = feeDemandData?.data?.data?.find(
    (item) => item.monthName == currentMonthFull
  );

  if (feeDemandData.isLoading) {
    return <StudentPortalSkeleton />;
  }

  console.log('the user image...', user);

  return (
    <Page title="Home">
      <CircularEventById
        id={getId!}
        isDialogOpen={isDialogOpen}
        setIsDialogOpen={setIsDialogOpen}
      />
      <SearchMenu />
      <div className="font-sans bg-slate-50 dark:bg-gray-900 min-h-screen pb-12 transition-colors duration-200">
        {/* Student Info Banner */}
        <div className="max-w-full mx-4 mt-4">
          {/* Main card with glass morphism effect */}
          <div className="bg-white/90 dark:bg-gray-800/90 backdrop-blur-sm rounded-2xl p-4 relative overflow-hidden shadow-lg border border-gray-100 dark:border-gray-700">
            {/* Status bar at top - like native apps */}
            <div className="top-0 left-0 right-0 flex justify-center items-center px-6 py-1 border-b border-gray-100 dark:border-gray-700 -mt-3">
              <div>
                <h1 className="font-bold text-lg text-primary dark:text-blue-400 text-center">
                  {localStorage.getItem('schoolName')}{' '}
                </h1>
              </div>
            </div>

            {/* Main content with padding from status bar */}
            <div className="pt-6 relative z-10">
              {/* Profile summary section */}
              <div className="flex items-center gap-4 py-2">
                <div className="bg-gray-100 dark:bg-gray-700 h-16 w-16 rounded-full flex items-center justify-center text-gray-500 dark:text-gray-400 font-bold text-lg">
                  {user?.imageUrl && (
                    <Image
                      src={`${BASE_URI}/${user?.imageUrl}`}
                      className="h-16 w-16 rounded-full border-2 border-gray-400 dark:border-gray-500"
                      alt="Profile Pic"
                    />
                  )}
                  {!user?.imageUrl && (
                    <Image
                      src={'/dummy-user.png'}
                      className="h-16 w-16 rounded-full border-2 border-gray-400 dark:border-gray-500"
                      alt="Profile Pic"
                    />
                  )}
                </div>
                <div className="flex-1">
                  <p className="text-xs font-bold text-gray-800 dark:text-gray-200">
                    Roll no: {user?.student?.rollNo ?? 'NA'}
                  </p>
                  <p className="text-xs font-semibold text-gray-500 dark:text-gray-400">
                    Class: {classData?.data?.data?.className ?? 'NA'} | Section:{' '}
                    {classData?.data?.data?.sectionName ?? 'NA'}
                  </p>
                  <span className="text-xs font-medium text-gray-500 dark:text-gray-400">
                    Session-: {user?.student?.session}
                  </span>
                </div>
                <Link to="/lms-app/profile" className="inline-flex">
                  <button className="flex items-center justify-center p-2 text-gray-700 dark:text-gray-300 transition-all duration-200 bg-white dark:bg-gray-800 rounded-full shadow-sm hover:bg-blue-50 dark:hover:bg-gray-700 hover:text-blue-600 dark:hover:text-blue-400 hover:shadow-md border border-gray-200 dark:border-gray-600 hover:border-blue-200 dark:hover:border-blue-500">
                    <Edit size={18} />
                  </button>
                </Link>
              </div>
            </div>
          </div>
        </div>

        {/* No Fee Records Card */}
        {feeAmountByStudentId?.data?.data?.totalFeeAmount! <= 0 &&
        feeAmountByStudentId?.data?.data?.totalPaidFeeAmount! <= 0 &&
        feeAmountByStudentId?.data?.data?.totalUnPaidFeeAmount! <= 0 ? (
          <div className="px-4 my-4">
            <Card className="border-none backdrop-blur-sm rounded-2xl shadow-lg bg-gradient-to-br from-blue-50 to-indigo-50 dark:from-blue-900/20 dark:to-indigo-900/20 lg:col-span-2">
              <CardContent className="p-6">
                <div className="flex flex-col items-center text-center mb-6">
                  <div className="h-16 w-16 rounded-full bg-blue-100 dark:bg-blue-900/50 flex items-center justify-center mb-4">
                    <FileText className="h-8 w-8 text-blue-500 dark:text-blue-400" />
                  </div>

                  <h3 className="text-xl font-bold text-blue-700 dark:text-blue-400 mb-1">
                    No Fee Records Found
                  </h3>
                </div>

                <div className="bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800 rounded-lg p-4 mb-6">
                  <div className="flex items-start gap-3">
                    <AlertCircle className="h-5 w-5 text-amber-500 dark:text-amber-400 mt-0.5 flex-shrink-0" />
                    <div>
                      <h4 className="text-sm font-medium text-amber-700 dark:text-amber-400 mb-1">
                        Contact School Administration
                      </h4>
                      <p className="text-xs text-amber-600 dark:text-amber-500">
                        Please contact school admin to get the updated fee
                        details.
                      </p>
                    </div>
                  </div>
                </div>
              </CardContent>
            </Card>
          </div>
        ) : (
          <div className="mx-4 my-4">
            <div className="rounded-2xl overflow-hidden border-none shadow-xl">
              <div
                className={`${
                  getIsPaidAmount?.isPaid
                    ? 'bg-gradient-to-br from-emerald-500 to-teal-600 dark:from-emerald-600 dark:to-teal-700'
                    : 'bg-gradient-to-br from-primary to-primary/90 dark:from-blue-600 dark:to-blue-700'
                } rounded-2xl p-6 relative overflow-hidden`}
              >
                {/* Decorative elements */}
                <div className="absolute top-0 right-0 w-40 h-40 bg-white/10 dark:bg-white/5 rounded-full -mr-16 -mt-16"></div>
                <div className="absolute bottom-0 left-0 w-28 h-28 bg-white/10 dark:bg-white/5 rounded-full -ml-12 -mb-12"></div>
                <div className="absolute top-1/4 left-1/3 w-12 h-12 bg-white/10 dark:bg-white/5 rounded-full blur-sm"></div>
                <div className="absolute bottom-1/4 right-1/3 w-16 h-16 bg-white/5 dark:bg-white/3 rounded-full blur-sm"></div>

                <div className="flex justify-between items-center mb-5">
                  <h3 className="font-bold text-white text-sm">Paid Upto</h3>
                  <div className="flex flex-col items-end">
                    {getIsPaidAmount?.isPaid ? (
                      <div className="flex items-center gap-2">
                        <div className="flex items-center gap-2 bg-white/20 dark:bg-white/10 px-4 py-1 rounded-full shadow-md backdrop-blur-sm">
                          <div className="w-3 h-3 bg-green-300 dark:bg-green-400 rounded-full animate-pulse"></div>
                          <span className="text-white font-bold text-xs">
                            {feeDemandData?.data?.paidMonthsRange}
                          </span>
                        </div>
                      </div>
                    ) : (
                      <div className="flex items-center gap-2">
                        <div className="flex items-center gap-2 bg-white/20 dark:bg-white/10 px-4 py-1 rounded-full shadow-md backdrop-blur-sm">
                          <div className="w-3 h-3 bg-red-300 dark:bg-red-400 rounded-full animate-pulse"></div>
                          <span className="text-white font-bold text-xs">
                            {feeDemandData?.data?.paidMonthsRange}
                          </span>
                        </div>
                      </div>
                    )}
                  </div>
                </div>

                <div className="flex items-center gap-5 mb-4">
                  <div
                    className={`w-16 h-16 ${
                      getIsPaidAmount?.isPaid
                        ? 'bg-emerald-50 dark:bg-emerald-900/50'
                        : 'bg-blue-50 dark:bg-blue-900/50'
                    } rounded-2xl flex items-center justify-center flex-shrink-0 shadow-lg`}
                  >
                    <svg
                      xmlns="http://www.w3.org/2000/svg"
                      className={`h-8 w-8 ${
                        getIsPaidAmount?.isPaid
                          ? 'text-emerald-600 dark:text-emerald-400'
                          : 'text-blue-600 dark:text-blue-400'
                      }`}
                      fill="none"
                      viewBox="0 0 24 24"
                      stroke="currentColor"
                    >
                      {getIsPaidAmount?.isPaid ? (
                        <path
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          strokeWidth={2}
                          d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z"
                        />
                      ) : (
                        <path
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          strokeWidth={2}
                          d="M17 9V7a2 2 0 00-2-2H5a2 2 0 00-2 2v6a2 2 0 002 2h2m2 4h10a2 2 0 002-2v-6a2 2 0 00-2-2H9a2 2 0 00-2 2v6a2 2 0 002 2zm7-5a2 2 0 11-4 0 2 2 0 014 0z"
                        />
                      )}
                    </svg>
                  </div>
                  <div>
                    <span
                      className={`${
                        getIsPaidAmount?.isPaid
                          ? 'text-emerald-100 dark:text-emerald-200'
                          : 'text-blue-100 dark:text-blue-200'
                      } text-sm font-medium block mb-1`}
                    >
                      {getIsPaidAmount?.isPaid
                        ? 'Due Amount'
                        : 'Due Fee Amount'}
                    </span>
                    <h3 className="text-white text-3xl font-extrabold">
                      ₹ {feeDemandData?.data?.pendingAmount || '0'}
                    </h3>
                  </div>
                </div>

                {getIsPaidAmount?.isPaid ? (
                  <div className="bg-white/20 dark:bg-white/10 p-2 rounded-xl mb-4 backdrop-blur-sm border border-white/30 dark:border-white/20 shadow-lg">
                    <div className="flex items-center gap-3">
                      <svg
                        xmlns="http://www.w3.org/2000/svg"
                        className="h-6 w-6 text-white"
                        viewBox="0 0 20 20"
                        fill="currentColor"
                      >
                        <path
                          fillRule="evenodd"
                          d="M10 18a8 8 0 100-16 8 8 0 000 16zm3.707-9.293a1 1 0 00-1.414-1.414L9 10.586 7.707 9.293a1 1 0 00-1.414 1.414l2 2a1 1 0 001.414 0l4-4z"
                          clipRule="evenodd"
                        />
                      </svg>
                      <p className="text-xs text-white font-medium">
                        Your fees have been paid successfully for the term.
                        Thank you!
                      </p>
                    </div>
                  </div>
                ) : (
                  <div className="bg-yellow-50 dark:bg-yellow-900/20 p-2 rounded-xl mb-4 border border-yellow-100 dark:border-yellow-800 shadow-sm">
                    <div className="flex items-start gap-3">
                      <svg
                        xmlns="http://www.w3.org/2000/svg"
                        className="h-6 w-6 text-yellow-500 dark:text-yellow-400 mt-0.5 flex-shrink-0"
                        viewBox="0 0 20 20"
                        fill="currentColor"
                      >
                        <path
                          fillRule="evenodd"
                          d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z"
                          clipRule="evenodd"
                        />
                      </svg>
                      <p className="text-xs text-yellow-700 dark:text-yellow-400 font-medium">
                        Please clear your pending fees before the due date to
                        avoid late payment charges.
                      </p>
                    </div>
                  </div>
                )}

                <Link to="/lms-app/student/payment" className="block w-full">
                  <Button
                    className={`w-full ${
                      getIsPaidAmount?.isPaid
                        ? 'bg-white hover:bg-emerald-50 dark:hover:bg-emerald-900/20 text-emerald-600 dark:text-emerald-400'
                        : 'bg-white hover:bg-blue-50 dark:hover:bg-blue-900/20 text-blue-600 dark:text-blue-400'
                    } font-bold text-xs rounded-xl py-4 shadow-lg transition-all duration-200 border-2 border-transparent hover:border-white/50 dark:hover:border-white/30`}
                  >
                    {getIsPaidAmount?.isPaid
                      ? 'VIEW PAYMENT HISTORY'
                      : 'VIEW FEE DETAILS'}
                  </Button>
                </Link>
              </div>
            </div>
          </div>
        )}

        {/* Menu Services */}
        <div className="px-4 mt-6">
          <h3 className="font-bold text-slate-800 dark:text-gray-200 text-base mb-3">
            School Services
          </h3>
          <div className="bg-white dark:bg-gray-800 rounded-2xl p-4 shadow-md border border-gray-100 dark:border-gray-700">
            <div className="grid grid-cols-4 gap-4">
              {menuItems
                ?.filter((item) => item.category == 'school')
                ?.map((itm, index) => {
                  return (
                    <Link key={index} to={itm.path}>
                      <div
                        className={`bg-primary/5 dark:bg-blue-500/10 rounded-xl p-2 flex flex-col items-center justify-center shadow-sm w-full h-[4.2rem] hover:bg-primary/10 dark:hover:bg-blue-500/20 transition-colors duration-200`}
                      >
                        <div className="flex justify-center text-2xl mb-1">
                          {itm?.isImage ? (
                            <>
                              <Image
                                src={itm.icon}
                                alt="Avatar"
                                className="overflow-hidden w-7"
                              />
                            </>
                          ) : (
                            <span className="text-primary dark:text-blue-400">
                              {itm.icon}
                            </span>
                          )}
                        </div>
                        <div className="text-xs font-semibold text-slate-700 dark:text-gray-300 text-center leading-tight w-20 truncate">
                          {i18n.t(itm.title)}
                        </div>
                      </div>
                    </Link>
                  );
                })}
            </div>
          </div>
        </div>

        {/* TOOLS */}
        <div className="px-4 mt-6">
          <h3 className="font-bold text-slate-800 dark:text-gray-200 text-base mb-3">
            Explore Tools
          </h3>
          <div className="bg-white dark:bg-gray-800 rounded-2xl p-4 shadow-md border border-gray-100 dark:border-gray-700">
            <div className="grid grid-cols-4 gap-4">
              {menuItems
                ?.filter((item) => item.category == 'tools')
                ?.map((itm, index) => {
                  return (
                    <Link key={index} to={itm.path}>
                      <div
                        className={`bg-primary/5 dark:bg-blue-500/10 rounded-xl p-2 flex flex-col items-center justify-center shadow-sm w-full h-[4.2rem] hover:bg-primary/10 dark:hover:bg-blue-500/20 transition-colors duration-200`}
                      >
                        <div className="flex justify-center text-2xl mb-1">
                          {itm?.isImage ? (
                            <>
                              <Image
                                src={itm.icon}
                                alt="Avatar"
                                className="overflow-hidden w-7"
                              />
                            </>
                          ) : (
                            <span className="text-primary dark:text-blue-400">
                              {itm.icon}
                            </span>
                          )}
                        </div>
                        <div className="text-xs font-semibold text-slate-700 dark:text-gray-300 text-center leading-tight">
                          {i18n.t(itm.title)}
                        </div>
                      </div>
                    </Link>
                  );
                })}
            </div>
          </div>
        </div>

        {/* Announcements */}
        <div className="px-4 mt-6">
          <div className="flex justify-between items-center mb-3">
            <h3 className="font-bold text-slate-800 dark:text-gray-200 text-base">
              Announcements
            </h3>
            <Link to="/lms-app/circular-events">
              <span className="text-blue-600 dark:text-blue-400 text-sm font-medium">
                View All
              </span>
            </Link>
          </div>
          <Card className="rounded-2xl border-none shadow-md overflow-hidden bg-white dark:bg-gray-800 border border-gray-100 dark:border-gray-700">
            {circularEventData?.data?.data?.docs?.length! > 0 ? (
              circularEventData?.data?.data?.docs?.map((item, index) => {
                return (
                  <div
                    className="p-4 border-b border-slate-100 dark:border-gray-700 hover:bg-gray-50 dark:hover:bg-gray-700/50 transition-colors duration-200 cursor-pointer"
                    onClick={() => {
                      setIsDialogOpen(true);
                      setId(item._id);
                    }}
                    key={index + 1}
                  >
                    <div className="flex gap-3">
                      <div className="w-10 h-10 bg-red-500 dark:bg-red-600 rounded-xl flex items-center justify-center flex-shrink-0">
                        <svg
                          xmlns="http://www.w3.org/2000/svg"
                          className="h-6 w-6 text-white"
                          fill="none"
                          viewBox="0 0 24 24"
                          stroke="currentColor"
                        >
                          <path
                            strokeLinecap="round"
                            strokeLinejoin="round"
                            strokeWidth={2}
                            d="M11 5.882V19.24a1.76 1.76 0 01-3.417.592l-2.147-6.15M18 13a3 3 0 100-6M5.436 13.683A4.001 4.001 0 017 6h1.832c4.1 0 7.625-1.234 9.168-3v14c-1.543-1.766-5.067-3-9.168-3H7a3.988 3.988 0 01-1.564-.317z"
                          />
                        </svg>
                      </div>
                      <div>
                        <h4 className="font-bold text-slate-800 dark:text-gray-200">
                          {item?.title}
                        </h4>
                        <p className="text-sm text-slate-500 dark:text-gray-400">
                          {item?.description?.length > 120
                            ? item?.description?.slice(0, 120) + '...'
                            : item?.description}
                        </p>
                        <div className="text-xs text-slate-400 dark:text-gray-500 mt-1">
                          {moment(item?.circularEventDate).format(
                            'MMMM D, YYYY'
                          )}
                        </div>
                      </div>
                    </div>
                  </div>
                );
              })
            ) : (
              <div className="p-8 text-center">
                <div className="mb-4 flex justify-center">
                  <div className="w-16 h-16 bg-slate-100 dark:bg-gray-700 rounded-full flex items-center justify-center">
                    <svg
                      xmlns="http://www.w3.org/2000/svg"
                      className="h-8 w-8 text-slate-400 dark:text-gray-500"
                      fill="none"
                      viewBox="0 0 24 24"
                      stroke="currentColor"
                    >
                      <path
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        strokeWidth={1.5}
                        d="M15 17h5l-1.405-1.405A2.032 2.032 0 0118 14.158V11a6.002 6.002 0 00-4-5.659V5a2 2 0 10-4 0v.341C7.67 6.165 6 8.388 6 11v3.159c0 .538-.214 1.055-.595 1.436L4 17h5m6 0v1a3 3 0 11-6 0v-1m6 0H9"
                      />
                    </svg>
                  </div>
                </div>
                <h4 className="font-medium text-slate-700 dark:text-gray-300 mb-1">
                  No Announcements
                </h4>
                <p className="text-slate-500 dark:text-gray-400 text-sm">
                  There are no announcements at this time. Check back later.
                </p>
              </div>
            )}
          </Card>
        </div>
      </div>
    </Page>
  );
}
